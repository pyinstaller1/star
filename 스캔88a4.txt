# 정렬하고자 하는 대상인 A[first]…A[last]를 
# 피벗(A[last]) 기준으로 분할하는 함수
def partition(A, first, last):
    p = A[last]
    low = first
    high = last
    while low < high:
        while p > A[low] and low < high: 
            low += 1
        while p <= A[high] and low < high: 
            high -= 1
        if low < high: 
            A[low], A[high] = A[high], A[low]
    A[low], A[last] = A[last], A[low]
    return low
